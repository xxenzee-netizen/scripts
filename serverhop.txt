-- Smooth Draggable Logic (Bound to TopBar)
local dragging, dragInput, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = input.Position; startPos = MainFrame.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
TopBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local isSearching = false

TeleportService.TeleportInitFailed:Connect(function()
    if isSearching then
        StatusLabel.Text = "Bypassing restricted server..."
        task.wait(1.5)
    end
end)

local function findAndHopServer(targetPlayers)
    -- Ascending order instantly checks empty/small servers first
    local url = "https://games.roblox.com/v1/games/" .. currentPlaceId .. "/servers/Public?sortOrder=Asc&excludeFullGames=true&limit=100"
    local cursor = ""

    while isSearching do
        local fetchUrl = (cursor == "") and url or url .. "&cursor=" .. cursor
        local success, result = pcall(function() return game:HttpGet(fetchUrl) end)
        
        if not success then
            StatusLabel.Text = "Network error, retrying..."
            task.wait(1)
            continue
        elseif result:match("TooManyRequests") then
            StatusLabel.Text = "Rate limited. Waiting 3s..."
            task.wait(3)
            continue
        end

        local parseSuccess, decodedData = pcall(function() return HttpService:JSONDecode(result) end)
        if parseSuccess and decodedData and decodedData.data then
            local foundServer = nil

            for _, server in ipairs(decodedData.data) do
                -- FILTER: Ping > 0 ensures it's not a dead ghost server. 
                -- FILTER: Ping < 300 & FPS > 15 ensures it's updating in real-time, preventing the "1 player but actually 4" bug!
                local isLiveServer = type(server.ping) == "number" and server.ping > 0 and server.ping < 300 and server.fps > 15
                local isNotCurrent = server.id ~= game.JobId

                if isLiveServer and isNotCurrent then
                    if server.playing == targetPlayers then
                        foundServer = server.id
                        break
                    end
                end
            end

            if foundServer then
                StatusLabel.Text = "Live Server Found! Joining..."
                StatusLabel.TextColor3 = Color3.fromRGB(0, 120, 0)
                
                pcall(function()
                    TeleportService:TeleportToPlaceInstance(currentPlaceId, foundServer, LocalPlayer)
                end)
                
                task.wait(6) 
                continue 
            end

            if decodedData.nextPageCursor then
                cursor = decodedData.nextPageCursor
                StatusLabel.Text = "Scanning next page..."
            else
                StatusLabel.Text = "No active servers match."
                break
            end
        else
            break
        end
    end

    isSearching = false
    HopButton.Text = "TELEPORT"
    HopButton.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
    StatusLabel.TextColor3 = Color3.fromRGB(30, 30, 30)
    if not StatusLabel.Text:match("Servers match") then
        StatusLabel.Text = "Search Stopped."
    end
end

HopButton.MouseButton1Click:Connect(function()
    if isSearching then
        isSearching = false
        HopButton.Text = "TELEPORT"
        HopButton.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
        StatusLabel.Text = "Search Cancelled."
        StatusLabel.TextColor3 = Color3.fromRGB(30, 30, 30)
        return
    end
    
    local targetAmount = tonumber(PlayerInput.Text)
    
    if targetAmount and targetAmount >= 0 then
        isSearching = true
        HopButton.Text = "STOP HOP"
        HopButton.BackgroundColor3 = Color3.fromRGB(200, 40, 40)
        StatusLabel.Text = "Searching for " .. targetAmount .. " players..."
        StatusLabel.TextColor3 = Color3.fromRGB(0, 80, 180)
        
        task.spawn(function()
            findAndHopServer(targetAmount)
        end)
    else
        StatusLabel.Text = "Enter a valid number (0+)."
        StatusLabel.TextColor3 = Color3.fromRGB(200, 0, 0)
    end
end)
